string SalesSummary(IEnumerable<string> salesFiles)
{
    double total = 0;
    List<string> details = new List<string>();

    foreach (var file in salesFiles)
    {

        //read contents of file
        string salesJson = File.ReadAllText(file);
        //parse contents as json 
        SalesData? data = JsonConvert.DeserializeObject<SalesData?>(salesJson);
        //add totals together in total var
        total += data?.Total ?? 0;

        //add file name and store total to list
        details.Add($"{Path.GetRelativePath(storesDirectory, file)}: {data?.Total ?? 0}");
    }

    StringBuilder report = new();

    report.AppendLine("Sales Summary");
    report.AppendLine("----------------------------");
    report.AppendLine($"Total Sales: {total:C}");
    report.AppendLine();
    report.AppendLine("Details:");

    foreach (string detail in details)
    {
        report.AppendLine(detail);
    }

    return report.ToString();
   }
